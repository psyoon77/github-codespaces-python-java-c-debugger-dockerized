2 vim config - add the following -
Host *
  AddKeysToAgent yes
  IdentitiesOnly yes

Host github-[name of the key]
  HostName github.com
  User git
  IdentityFile ~/.ssh/id_ed25519_[name of the key]

3 start ssh agent
eval "$(ssh-agent -s)"

4 add key to ssh agent
ssh-add ~/.ssh/id_ed25519_[name of the key]

5 copy the public ssh key
cat ~/.ssh/id_ed25519_[name of the key].pub

6 upload the public ssh key on the github account
settings - ssh and gpg keys - ssh keys - new ssh key -
	name it [name of the key recommended] - paste the public key

7 .ssh folder, ssh keys, and config file privilege set
